\topic{Number Theory Facts}
{\small
\textbf{Euler.} $a^{\varphi(m)} \equiv 1$ if $\gcd(a, m) = 1$. For any $a$ and $e \ge \log_2 m$: $a^e \equiv a^{(e \bmod \varphi(m)) + \varphi(m)}$. \textbf{Wilson.} $(p-1)! \equiv -1 \pmod p$.
\textbf{M\"obius.} $g(n) = \sum_{d \mid n} f(d) \iff f(n) = \sum_{d \mid n} \mu(d) g(n/d)$. $\sum_{d \mid n} \mu(d) = [n = 1]$, $\sum_{d \mid n} \varphi(d) = n$, $\sum_{i \le n} \gcd(i, n) = \sum_{d \mid n} d\,\varphi(n/d)$. $\#\{(a, b) \le n : \gcd = 1\} = \sum_d \mu(d) \lfloor n/d \rfloor^2$.
\textbf{Diophantine.} $ax + by = c$ solvable iff $g = \gcd(a, b) \mid c$: $x = x_0 c/g + k b/g$, $y = y_0 c/g - k a/g$. Largest non-representable with $\gcd(a, b) = 1$: $ab - a - b$.
\textbf{Legendre.} $v_p(n!) = \sum_i \lfloor n/p^i \rfloor$. \textbf{Kummer.} $v_p\binom{n}{k}$ = carries when adding $k$ and $n - k$ in base $p$.
\textbf{Primitive roots} exist mod $1, 2, 4, p^k, 2p^k$. \textbf{Pythagorean triples}: $k(m^2 - n^2), 2kmn, k(m^2 + n^2)$, $m > n$ coprime, not both odd.
\textbf{Divisors.} Max count: 1344 for $n \le 10^9$, 103680 for $n \le 10^{18}$.
\textbf{NTT primes.} $998244353 = 119 \cdot 2^{23} + 1$, $167772161 = 5 \cdot 2^{25} + 1$, $469762049 = 7 \cdot 2^{26} + 1$, all with root 3. Other primes: $10^9 + 7$, $10^9 + 9$, $10^{18} + 3$.
\par}
