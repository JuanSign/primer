\topic{Math Formulas}
{\small
\textbf{Sums.} $\sum i = \frac{n(n+1)}{2}$, $\sum i^2 = \frac{n(n+1)(2n+1)}{6}$, $\sum i^3 = \frac{n^2(n+1)^2}{4}$, $\sum_{i=0}^{n} x^i = \frac{x^{n+1} - 1}{x - 1}$, $\sum_{i \ge 1} i x^{i} = \frac{x}{(1-x)^2}$ for $|x| < 1$. $H_n \approx \ln n + 0.5772$.
\textbf{Trig.} $\sin(a \pm b) = \sin a\cos b \pm \cos a\sin b$, $\cos(a \pm b) = \cos a\cos b \mp \sin a\sin b$, $\tan(a + b) = \frac{\tan a + \tan b}{1 - \tan a\tan b}$, $\sin a + \sin b = 2\sin\frac{a+b}{2}\cos\frac{a-b}{2}$, $\cos a + \cos b = 2\cos\frac{a+b}{2}\cos\frac{a-b}{2}$.
\textbf{Quadratic.} $q = -\frac12(b + \mathrm{sgn}(b)\sqrt{b^2 - 4ac})$, roots $q/a$ and $c/q$ (no cancellation).
\textbf{Calculus.} $(\arctan x)' = \frac{1}{1+x^2}$, $(\arcsin x)' = \frac{1}{\sqrt{1-x^2}}$, $\int \ln x = x\ln x - x$, $\int x e^{ax} = \frac{e^{ax}}{a^2}(ax - 1)$, $\int_0^\infty e^{-x^2} = \frac{\sqrt\pi}{2}$.
\textbf{Probability.} $E[X + Y] = E[X] + E[Y]$, $\mathrm{Var}(X) = E[X^2] - E[X]^2$, $\mathrm{Var}(aX + b) = a^2\mathrm{Var}(X)$. Binomial$(n, p)$: $np$, $np(1-p)$. Geometric (trials to first success): $1/p$, $(1-p)/p^2$. Poisson$(\lambda)$: $\lambda$, $\lambda$. Uniform$[a, b]$: $\frac{a+b}{2}$, $\frac{(b-a)^2}{12}$. Exponential$(\lambda)$: $1/\lambda$, $1/\lambda^2$. Coupon collector: $nH_n$. Markov chain: absorption probabilities and expected steps solve a linear system.
\par}
