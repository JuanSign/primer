\topic{Formulas}
{\small
\textbf{Triangle.} Area $= \frac12 ab\sin C = \sqrt{s(s-a)(s-b)(s-c)}$, $r = A/s$, $R = \frac{abc}{4A}$, $\frac{a}{\sin A} = 2R$, $c^2 = a^2 + b^2 - 2ab\cos C$. Median $m_a = \frac12\sqrt{2b^2 + 2c^2 - a^2}$.
\textbf{Pick.} $A = I + B/2 - 1$. \textbf{Euler.} $V - E + F = 2$ (planar, connected).
\textbf{Sphere.} $V = \frac43\pi r^3$, $S = 4\pi r^2$. Cap of height $h$: $V = \frac{\pi h^2 (3r - h)}{3}$, $S = 2\pi r h$. Cone $V = \frac13\pi r^2 h$. Great circle: $d = r\arccos(\sin\varphi_1\sin\varphi_2 + \cos\varphi_1\cos\varphi_2\cos\Delta\lambda)$.
\textbf{Rotation} by $\theta$: $(x\cos\theta - y\sin\theta,\ x\sin\theta + y\cos\theta)$. \textbf{Manhattan to Chebyshev}: $(x + y, x - y)$.
\par}
